208    def enrich_spans_for_export(self, spans: list[ReadableSpan]) -> None:
209        """Deferred enrichment, turn-attr copy, then GenAI/voice rename for export."""
210        self._turns.correct_function_tool_timing(spans)
211
212        llm_spans = sorted(
213            (s for s in spans if s.name == "llm_node"),
214            key=lambda s: s.end_time or 0,
215        )
216        for span in llm_spans:
217            # FIFO token attach before stamping active_speech_id (which would
218            # pin every span to the latest speech and break multi-node batches).
219            self._tokens.apply_plugin_llm_usage_to_span(
220                span, dict(span.attributes or {}), prefer_fifo=True
221            )
222            attrs = span.attributes or {}
223            if not attrs.get(ATTR_LK_SPEECH_ID):
224                bootstrap = get_job_bootstrap()
225                speech_id = ""
226                if bootstrap is not None:
227                    speech_id = bootstrap.state.active_speech_id.strip()
228                if speech_id:
229                    self._set(span, ATTR_LK_SPEECH_ID, speech_id)
230
231        self._turns.merge_native_turn_attrs_onto_parlot_turns(spans)
232
233        from ._telemetry_compare import compare_enabled, get_compare_logger
234
235        if compare_enabled():
236            bootstrap = get_job_bootstrap()
237            if bootstrap is not None and bootstrap.state.parlot_session_id:
238                session_id = bootstrap.state.parlot_session_id
239                for span in spans:
240                    if span.name == "llm_node":
241                        get_compare_logger().accumulate_export_tokens(
242                            session_id,
243                            span_name=span.name or "",
244                            attrs=dict(span.attributes or {}),
245                        )
246
247        # Rename LiveKit-native ops → GenAI/voice names before export filter.
248        for span in spans:
249            native = span.name or ""
250            if native in NATIVE_TURN_SPANS:
251                # Dropped by export filter (remap returns None); attrs already merged.
252                continue
253            if remap_livekit_span_name(native, span.attributes or {}) is not None:
254                apply_livekit_span_rename(span)
255                role = livekit_agent_role_for_span(span.name or "")
256                if role and not (span.attributes or {}).get(ATTR_AGENT_ROLE):
257                    self._set(span, ATTR_AGENT_ROLE, role)

367    def _resolve_session_state(
368        self, span: ReadableSpan, attrs: Mapping[str, AttributeValue]
369    ) -> _LiveKitSessionState | None:
370        bootstrap = get_job_bootstrap()
371        if bootstrap is not None:
372            return bootstrap.state
373        sid = attrs.get(ATTR_SESSION_ID)
374        if sid:
375            state = self._sessions.get(str(sid))
376            if state is not None:
377                return state
378        job_id = attrs.get(ATTR_LK_JOB_ID) or attrs.get(METADATA_JOB_ID)
379        if job_id:
380            state = self._sessions.get(str(job_id))
381            if state is not None:
382                return state
383        name = span.name or ""
384        if (
385            name in _AGENT_PIPELINE_SPANS
386            or name in _AGENT_LABEL_SPANS
387            or job_id is not None
388        ):
389            logger.debug(
390                "parlot: no session bootstrap for span %s (trace=%s); skipping enrich",
391                name,
392                self._trace_id_hex(span),
393            )
394            return None
395        logger.error(
396            "parlot: no session bootstrap for span %s (trace=%s)",
397            name,

402    def _track_agent_label(
403        self, state: _LiveKitSessionState, attrs: Mapping[str, AttributeValue]
404    ) -> None:
405        raw_label = attrs.get(ATTR_LK_AGENT_LABEL)

416    def _active_agent_id(
417        self,
418        state: _LiveKitSessionState,
419        attrs: Mapping[str, AttributeValue],
420        *,
421        label_override: str | None = None,
422    ) -> str:
423        if label_override:
424            override = topology_agent_name(label_override)
425            if override:
426                return override
427        # Prefer sub-agent labels over worker name; never use LiveKit AD_* dispatch ids.
428        for candidate in (
